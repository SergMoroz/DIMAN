// Хэш снимка для детерминированности анализа
function calculateQuickHash(str) {
  let hash = 0;
  for (let i = 0; i < Math.min(str.length, 6000); i++) {
    const char = str.charCodeAt(i);
    hash = ((hash << 5) - hash) + char;
    hash |= 0;
  }
  return Math.abs(hash).toString(16).toUpperCase();
}

            <div className="flex items-center justify-between mb-1.5">
              <span className="text-[11px] font-bold text-slate-200 flex items-center gap-1.5">
                <Gauge className="w-3.5 h-3.5 text-cyan-400" />
                Шероховатость чертежа (ГОСТ 2789-73):
              </span>
              <span className="text-[11px] font-mono font-bold text-cyan-300 bg-cyan-950/60 px-2 py-0.2 rounded border border-cyan-800/60">
                {targetRoughness}
              </span>
            </div>

            <div className="flex gap-1.5 overflow-x-auto pb-0.5 text-[10px] font-mono scrollbar-none">
              {ROUGHNESS_PRESETS.map((opt) => (
                <button
                  key={opt.value}
                  onClick={() => { triggerHaptic([10]); setTargetRoughness(opt.value); }}
                  className={`px-2 py-1 rounded-lg border whitespace-nowrap transition active:scale-95 ${
                    targetRoughness === opt.value
                      ? 'bg-gradient-to-r from-cyan-500 to-blue-600 text-slate-950 font-black border-cyan-400 shadow-sm'
                      : 'bg-slate-950 border-slate-800 text-slate-400 hover:border-slate-700'
                  }`}
                >
                  {opt.label}
                </button>
              ))}
            </div>
          </div>
        )}

        {/* Видоискатель детали с прецизионными рамками */}
        <div className="bg-slate-900 border border-slate-800 rounded-2xl overflow-hidden shadow-xl flex flex-col">
          
          <div className="bg-slate-950 px-2.5 py-1.5 border-b border-slate-800 flex items-center justify-between text-[11px]">
            <div className="flex items-center gap-1">
              <button 
                onClick={() => { triggerHaptic([10]); setZoomLevel(prev => Math.min(prev + 0.25, 2.5)); }}
                className="p-1 rounded-lg bg-slate-900 hover:bg-slate-800 text-slate-300 active:scale-90"
                title="Приблизить"
              >
                <ZoomIn className="w-3.5 h-3.5" />
              </button>
              <button 
                onClick={() => { triggerHaptic([10]); setZoomLevel(prev => Math.max(prev - 0.25, 1)); }}
                className="p-1 rounded-lg bg-slate-900 hover:bg-slate-800 text-slate-300 active:scale-90"
                title="Отдалить"
              >
                <ZoomOut className="w-3.5 h-3.5" />
              </button>
              <button 
                onClick={resetViewport}
                className="p-1 rounded-lg bg-slate-900 hover:bg-slate-800 text-slate-400 text-[10px] font-mono"
                title="Сброс масштаба"
              >
                1:1
              </button>
              <button 
                onClick={() => { triggerHaptic([10]); setShowGrid(!showGrid); }}
                className={`p-1 rounded-lg transition ${showGrid ? 'bg-cyan-500/20 text-cyan-400' : 'bg-slate-900 text-slate-400'}`}
                title="Координатная сетка"
              >
                <Layers className="w-3.5 h-3.5" />
              </button>
              <button 
                onClick={() => { triggerHaptic([10]); setMarkerStyle(prev => prev === 'hud' ? 'pin' : 'hud'); }}
                className={`p-1 rounded-lg text-[9px] font-mono transition ${markerStyle === 'hud' ? 'bg-cyan-500/20 text-cyan-300 border border-cyan-500/40' : 'bg-slate-900 text-slate-400'}`}
                title="Стиль рамок"
              >
                {markerStyle === 'hud' ? 'РАМКИ' : 'МЕТКИ'}
              </button>
            </div>

            <div className="flex gap-1">
              {[
                { id: 'normal', label: 'Норм' },
                { id: 'contrast', label: 'Контраст' },
                { id: 'mono', label: 'Ч/Б' },
                { id: 'invert', label: 'Инверс' }
              ].map(f => (
                <button
                  key={f.id}
                  onClick={() => { triggerHaptic([10]); setActiveFilter(f.id); }}
                  className={`px-1.5 py-0.5 rounded-md text-[9px] font-mono transition ${
                    activeFilter === f.id ? 'bg-cyan-400 text-slate-950 font-bold' : 'bg-slate-900 text-slate-400'
                  }`}
                >
                  {f.label}
                </button>
              ))}
            </div>
          </div>

          <div 
            ref={viewportContainerRef}
            className="relative aspect-square w-full bg-slate-950 flex items-center justify-center overflow-hidden"
          >
            {selectedImage ? (
              <div 
                className="relative flex items-center justify-center transition-transform duration-200 ease-out"
                style={{ 
                  transform: `scale(${zoomLevel}) translate(${panOffset.x}%, ${panOffset.y}%)`,
                  transformOrigin: 'center center'
                }}
              >
                <img
                  ref={imageElementRef}
                  src={selectedImage}
                  alt="Деталь"
                  className={`max-h-[360px] max-w-[360px] object-contain block ${getFilterClass()}`}
                />

                {showGrid && (
                  <div className="absolute inset-0 pointer-events-none bg-[linear-gradient(to_right,#06b6d420_1px,transparent_1px),linear-gradient(to_bottom,#06b6d420_1px,transparent_1px)] bg-[size:24px_24px] border border-cyan-500/30"></div>
                )}

                {/* Нанесение дефектов с Bounding Box */}
                {report?.detectedDefects?.map((item, idx) => {
                  const isSelected = selectedDefectId === item.id;
                  const isUnacceptable = item.admissibility === 'UNACCEPTABLE';
                  const isConditional = item.admissibility === 'CONDITIONAL';
                  const box = item.box || { cx: 50, cy: 50, w: 10, h: 10 };

                  const colorClass = isUnacceptable 
                    ? 'border-rose-500 text-rose-400 bg-rose-500/20' 
                    : isConditional 
                    ? 'border-amber-500 text-amber-400 bg-amber-500/20' 
                    : 'border-emerald-500 text-emerald-400 bg-emerald-500/20';

                  return (
                    <div
                      key={item.id}
                      onClick={() => focusOnDefect(item)}
                      style={{
                        position: 'absolute',
                        left: `${box.cx}%`,
                        top: `${box.cy}%`,
                        width: markerStyle === 'hud' ? `${Math.max(box.w, 7)}%` : '20px',
                        height: markerStyle === 'hud' ? `${Math.max(box.h, 7)}%` : '20px',
                        transform: 'translate(-50%, -50%)',
                        zIndex: isSelected ? 35 : 20
                      }}
                      className="cursor-pointer select-none transition-all duration-150"
                    >
                      {markerStyle === 'hud' ? (
                        <div className={`w-full h-full relative border-2 ${colorClass} rounded-sm shadow-lg flex items-center justify-center transition-transform ${isSelected ? 'scale-110 ring-2 ring-white/80' : 'hover:scale-105'}`}>
                          <div className="absolute -top-1 -left-1 w-1.5 h-1.5 border-t-2 border-l-2 border-white"></div>
                          <div className="absolute -top-1 -right-1 w-1.5 h-1.5 border-t-2 border-r-2 border-white"></div>
                          <div className="absolute -bottom-1 -left-1 w-1.5 h-1.5 border-b-2 border-l-2 border-white"></div>
                          <div className="absolute -bottom-1 -right-1 w-1.5 h-1.5 border-b-2 border-r-2 border-white"></div>
                          <div className="w-1.5 h-1.5 rounded-full bg-white shadow-sm"></div>
                          <div className="absolute -top-4 left-0 bg-slate-950/90 text-white font-mono text-[8px] px-1 py-0.2 rounded border border-slate-700 shadow font-bold whitespace-nowrap">
                            #{idx + 1} {item.isTolerableFlaw ? '[ДОПУСТИМО]' : item.defectName.slice(0, 9)}
                          </div>
                        </div>
                      ) : (
                        <div className="relative flex items-center justify-center w-full h-full">
                          <span className={`animate-ping absolute inline-flex h-full w-full rounded-full opacity-70 ${isUnacceptable ? 'bg-rose-500' : isConditional ? 'bg-amber-500' : 'bg-emerald-500'}`} />
                          <span className={`relative inline-flex rounded-full h-4 w-4 border-2 border-white items-center justify-center text-[8px] font-black text-white shadow-md ${
                            isUnacceptable ? 'bg-rose-600' : isConditional ? 'bg-amber-600' : 'bg-emerald-600'
                          }`}>
                            {idx + 1}
                          </span>
                        </div>
                      )}
                    </div>
                  );
                })}
              </div>
            ) : (
              <div className="flex flex-col items-center justify-center text-center p-6 text-slate-500">
                <Camera className="w-10 h-10 text-cyan-500/60 mb-2" />
                <span className="text-xs font-bold text-slate-300">Деталь не загружена</span>
                <span className="text-[10px] text-slate-500 mt-0.5">
                  {isAutoMode ? "Сделайте снимок на камеру OnePlus 13 для автозапуска" : "Сделайте снимок или выберите файл из галереи"}
                </span>
              </div>
            )}

            {isAnalyzing && (
              <div className="absolute inset-0 bg-slate-950/90 backdrop-blur-md flex flex-col items-center justify-center p-4 text-center z-30">
                <RefreshCw className="w-10 h-10 text-cyan-400 animate-spin mb-3" />
                <span className="text-xs font-mono font-bold text-cyan-300 uppercase tracking-widest">
                  {isAutoMode ? "ДИМАН: Автоопределение качества..." : "ДИМАН: Глубокая экспертиза..."}
                </span>
                <span className="text-[10px] text-slate-400 mt-1">ОТК фирмы «Полёт» • Фильтрация допустимого брака</span>
              </div>
            )}

            {/* Карточка выбранного дефекта */}
            {selectedDefectId && report && (
              (() => {
                const item = report.detectedDefects.find(d => d.id === selectedDefectId);
                if (!item) return null;
                return (
                  <div className="absolute bottom-2 inset-x-2 bg-slate-900/95 border border-cyan-500/50 p-2.5 rounded-xl shadow-2xl z-30 text-[11px] backdrop-blur max-h-52 overflow-y-auto">
                    <div className="flex items-center justify-between font-bold text-white mb-1">
                      <span className="flex items-center gap-1.5 text-cyan-400">
                        <Focus className="w-3.5 h-3.5 text-cyan-400" />
                        {item.defectName}
                      </span>
                      <button onClick={() => { setSelectedDefectId(null); resetViewport(); }} className="text-slate-400 p-0.5">✕</button>
                    </div>
                    <div className="text-[9px] font-mono mb-1 flex items-center justify-between">
                      <span className="text-cyan-300">{item.applicableGost}</span>
                      <span className={`font-bold ${
                        item.admissibility === 'UNACCEPTABLE' ? 'text-rose-400' :
                        item.admissibility === 'CONDITIONAL' ? 'text-amber-400' : 'text-emerald-400'
                      }`}>
                        {item.admissibilityLabel}
                      </span>
                    </div>
                    <div className="text-slate-300">
                      <span className="text-slate-400 font-medium">Зона:</span> {item.exactLocation}
                    </div>
                    <div className="text-slate-300 mt-0.5">
                      <span className="text-slate-400 font-medium">Оптический признак:</span> {item.physicalMorphology}
                    </div>
                    <div className="text-slate-200 mt-1 p-1.5 bg-slate-950 rounded border border-slate-800 text-[10px]">
                      <span className="text-cyan-400 font-bold block mb-0.5">Норма допуска:</span>
                      {item.toleranceRationale}
                    </div>
                    <div className="text-emerald-400 mt-1 pt-1 border-t border-slate-800 text-[10px]">
                      <span className="font-bold">Действие цеха:</span> {item.shopfloorAction}
                    </div>
                  </div>
                );
              })()
            )}
          </div>

          <div className="p-2 bg-slate-900 border-t border-slate-800 grid grid-cols-2 gap-2">
            <button
              onClick={() => cameraInputRef.current?.click()}
              className="py-2.5 px-3 rounded-xl bg-slate-800 active:bg-slate-700 text-slate-200 text-xs font-bold border border-slate-700 flex items-center justify-center gap-1.5 transition active:scale-95 shadow-sm"
            >
              <Camera className="w-4 h-4 text-cyan-400" />
              Снимок камеры
            </button>
            <button
              onClick={() => fileInputRef.current?.click()}
              className="py-2.5 px-3 rounded-xl bg-slate-800 active:bg-slate-700 text-slate-200 text-xs font-bold border border-slate-700 flex items-center justify-center gap-1.5 transition active:scale-95 shadow-sm"
            >
              <Upload className="w-4 h-4 text-cyan-400" />
              Из галереи
            </button>
          </div>
        </div>

        {imageSignature && (
          <div className="text-[9px] font-mono text-slate-500 px-1 flex justify-between">
            <span>Кадр: {imageSignature}</span>
            <span>Режим: {isAutoMode ? "АВТО-ОТК" : "РУЧНОЙ"} (T=0.0)</span>
          </div>
        )}

        {errorMsg && (
          <div className="p-2.5 rounded-xl bg-rose-950/60 border border-rose-800 text-rose-300 text-[11px] flex items-center gap-2">
            <AlertTriangle className="w-4 h-4 shrink-0 text-rose-400" />
            <span>{errorMsg}</span>
          </div>
        )}

        {/* РЕЗУЛЬТАТЫ ЭКСПЕРТИЗЫ */}
        {report && (
          <div className="flex flex-col gap-2.5 animate-in fade-in duration-200">
            
            {/* Карточка вердикта */}
            <div className={`p-3.5 rounded-xl border ${
              report.executiveVerdict.decision === 'APPROVED'
                ? 'bg-slate-900 border-emerald-500/60 text-emerald-400'
                : report.executiveVerdict.decision === 'CONDITIONAL'
                ? 'bg-slate-900 border-amber-500/60 text-amber-400'
                : 'bg-slate-900 border-rose-500/60 text-rose-400'
            }`}>
              <div className="flex items-center justify-between mb-1.5">
                <div className="flex items-center gap-2">
                  {report.executiveVerdict.decision === 'APPROVED' && <CheckCircle2 className="w-5 h-5 text-emerald-400" />}
                  {report.executiveVerdict.decision === 'CONDITIONAL' && <AlertTriangle className="w-5 h-5 text-amber-400" />}
                  {report.executiveVerdict.decision === 'REJECTED' && <AlertOctagon className="w-5 h-5 text-rose-400" />}
                  
                  <div>
                    <span className="text-[9px] font-mono uppercase text-slate-400 block">
                      ВЕРДИКТ (АО «ПОЛЁТ»):
                    </span>
                    <span className="text-xs font-black text-white uppercase">
                      {report.executiveVerdict.decision === 'APPROVED' ? 'ДОПУСК В СЕРИЮ (ГОДНО)' :
                       report.executiveVerdict.decision === 'CONDITIONAL' ? 'ДОПУСК С ДОРАБОТКОЙ В ЦЕХЕ' :
                       'ЗАПРЕТ ЗАПУСКА (БРАК)'}
                    </span>
                  </div>
                </div>
                <div className="text-right font-mono font-black text-lg text-white">
                  {report.executiveVerdict.overallScore}<span className="text-[10px] text-slate-400">/100</span>
                </div>
              </div>

              {/* Метрики АВТОПИЛОТА */}
              {report.autoPilotMetrics && (
                <div className="grid grid-cols-2 gap-1.5 my-2 p-2 bg-slate-950/90 rounded-lg border border-cyan-500/20 text-[10px] font-mono">
                  <div>
                    <span className="text-slate-500 block text-[8px]">ТЕХПРОЦЕСС (АВТО):</span>
                    <span className="text-slate-200 font-bold">{report.autoPilotMetrics.detectedProcessType}</span>
                  </div>
                  <div>
                    <span className="text-slate-500 block text-[8px]">КВАЛИТЕТ ЕСДП (АВТО):</span>
                    <span className="text-cyan-300 font-bold">{report.autoPilotMetrics.inferredAccuracyGrade}</span>
                  </div>
                  <div className="pt-1 border-t border-slate-800">
                    <span className="text-slate-500 block text-[8px]">ПРОГНОЗ БРАКА:</span>
                    <span className="text-amber-400 font-bold">{report.autoPilotMetrics.forecastDppm} DPPM</span>
                  </div>
                  <div className="pt-1 border-t border-slate-800">
                    <span className="text-slate-500 block text-[8px]">ТАКТ КОНТРОЛЯ:</span>
                    <span className="text-emerald-400 font-bold">{report.autoPilotMetrics.recommendedInspectionCycleSec} сек</span>
                  </div>
                </div>
              )}

              {/* Сводка дефектов */}
              <div className="grid grid-cols-3 gap-1 my-1.5 pt-1.5 border-t border-slate-800 text-center font-mono text-[9px]">
                <div className="bg-rose-950/40 border border-rose-800/60 p-1 rounded-lg text-rose-300">
                  <span className="block text-slate-400 text-[8px]">Брак (списание):</span>
                  <span className="font-bold text-[11px]">{report.executiveVerdict.unacceptableCount || 0}</span>
                </div>
                <div className="bg-amber-950/40 border border-amber-800/60 p-1 rounded-lg text-amber-300">
                  <span className="block text-slate-400 text-[8px]">Исправимый:</span>
                  <span className="font-bold text-[11px]">{report.executiveVerdict.conditionalCount || 0}</span>
                </div>
                <div className="bg-emerald-950/40 border border-emerald-800/60 p-1 rounded-lg text-emerald-300">
                  <span className="block text-slate-400 text-[8px]">Допустимый брак:</span>
                  <span className="font-bold text-[11px]">{report.executiveVerdict.acceptableCount || 0}</span>
                </div>
              </div>

              {/* Обоснование допустимого брака */}
              <div className="bg-slate-950 p-2 rounded-lg border border-slate-800 text-[10px] space-y-1 my-2">
                <div className="flex items-center gap-1.5 text-cyan-400 font-bold">
                  <ShieldCheck className="w-3.5 h-3.5" />
                  Оценка технологических допусков:
                </div>
                <div className="text-slate-300 leading-snug">
                  {report.executiveVerdict.toleranceAnalysisSummary}
                </div>
                <div className="text-emerald-400 font-mono text-[9px] pt-1 border-t border-slate-800">
                  Указание цеху: {report.executiveVerdict.shopfloorDecision}
                  {report.executiveVerdict.reworkEstimateMinutes > 0 && ` (~${report.executiveVerdict.reworkEstimateMinutes} мин)`}
                </div>
              </div>

              <p className="text-[11px] text-slate-300 bg-slate-950 p-2.5 rounded-xl border border-slate-800 leading-relaxed whitespace-pre-line">
                {report.executiveVerdict.summaryConclusion}
              </p>
            </div>

            {/* Вкладки */}
            <div className="flex rounded-xl bg-slate-900 p-1 border border-slate-800 text-[10px] font-bold overflow-x-auto scrollbar-none">
              <button 
                onClick={() => { triggerHaptic([10]); setActiveTab('verdict'); }}
                className={`py-1.5 px-2.5 rounded-lg whitespace-nowrap ${activeTab === 'verdict' ? 'bg-cyan-500/10 text-cyan-400 border border-cyan-500/30' : 'text-slate-400'}`}
              >
                Дефекты ({report.detectedDefects?.length || 0})
              </button>
              <button 
                onClick={() => { triggerHaptic([10]); setActiveTab('tolerance_map'); }}
                className={`py-1.5 px-2.5 rounded-lg whitespace-nowrap ${activeTab === 'tolerance_map' ? 'bg-cyan-500/10 text-cyan-400 border border-cyan-500/30' : 'text-slate-400'}`}
              >
                Допустимый брак
              </button>
              <button 
                onClick={() => { triggerHaptic([10]); setActiveTab('micro'); }}
                className={`py-1.5 px-2.5 rounded-lg whitespace-nowrap ${activeTab === 'micro' ? 'bg-cyan-500/10 text-cyan-400 border border-cyan-500/30' : 'text-slate-400'}`}
              >
                Микропрофиль
              </button>
              <button 
                onClick={() => { triggerHaptic([10]); setActiveTab('fmea'); }}
                className={`py-1.5 px-2.5 rounded-lg whitespace-nowrap ${activeTab === 'fmea' ? 'bg-cyan-500/10 text-cyan-400 border border-cyan-500/30' : 'text-slate-400'}`}
              >
                FMEA-риски
              </button>
            </div>

            {/* ВКЛАДКА 1: Дефекты */}
            {activeTab === 'verdict' && (
              <div className="flex flex-col gap-1.5">
                <div className="flex gap-1 overflow-x-auto pb-0.5 text-[9px] font-mono">
                  {[
                    { id: 'all', label: 'Все' },
                    { id: 'unacceptable', label: 'Недопустимые' },
                    { id: 'conditional', label: 'Условные' },
                    { id: 'acceptable', label: 'Допустимый брак' }
                  ].map(btn => (
                    <button
                      key={btn.id}
                      onClick={() => { triggerHaptic([10]); setDefectFilter(btn.id); }}
                      className={`px-2 py-0.5 rounded-md border whitespace-nowrap transition ${
                        defectFilter === btn.id 
                          ? 'bg-slate-800 border-cyan-500 text-cyan-300 font-bold' 
                          : 'bg-slate-900 border-slate-800 text-slate-400'
                      }`}
                    >
                      {btn.label}
                    </button>
                  ))}
                </div>

                {filteredDefects.length === 0 ? (
                  <div className="p-4 rounded-xl bg-slate-900 border border-slate-800 text-center text-xs text-slate-400">
                    <Check className="w-5 h-5 text-emerald-400 mx-auto mb-1" />
                    В выбранной категории отклонений не зафиксировано.
                  </div>
                ) : (
                  filteredDefects.map((d, idx) => (
                    <div 
                      key={d.id} 
                      onClick={() => focusOnDefect(d)}
                      className={`bg-slate-900 border rounded-xl p-2.5 text-xs transition cursor-pointer ${
                        selectedDefectId === d.id ? 'border-cyan-400 ring-2 ring-cyan-500/30 bg-slate-850' : 'border-slate-800'
                      }`}
                    >
                      <div className="flex justify-between items-start mb-1">
                        <span className="font-bold text-white flex items-center gap-1.5 text-[11px]">
                          <span className="w-4 h-4 rounded-full bg-slate-800 text-cyan-400 border border-slate-700 flex items-center justify-center text-[9px] font-mono font-bold">
                            {idx + 1}
                          </span>
                          <span className={`w-2 h-2 rounded-full ${d.admissibility === 'UNACCEPTABLE' ? 'bg-rose-500' : d.admissibility === 'CONDITIONAL' ? 'bg-amber-500' : 'bg-emerald-500'}`} />
                          {d.defectName}
                        </span>
                        <div className="flex items-center gap-1.5">
                          <span className="text-[9px] font-mono text-cyan-400 font-bold">{d.applicableGost}</span>
                          <Focus className="w-3 h-3 text-slate-500 hover:text-cyan-400" />
                        </div>
                      </div>
                      <div className="bg-slate-950 p-2 rounded-lg text-[10px] text-slate-300 space-y-0.5">
                        <div><span className="text-slate-500 font-medium">Зона:</span> {d.exactLocation}</div>
                        <div><span className="text-slate-500 font-medium">Признак:</span> {d.physicalMorphology}</div>
                        <div><span className="text-amber-400 font-medium">Риск:</span> {d.failureRisk}</div>
                        <div className="text-emerald-400 pt-0.5 border-t border-slate-800"><span className="font-bold">Цех:</span> {d.shopfloorAction}</div>
                      </div>
                    </div>
                  ))
                )}
              </div>
            )}

            {/* ВКЛАДКА 2: Допустимый брак */}
            {activeTab === 'tolerance_map' && (
              <div className="flex flex-col gap-1.5">
                <div className="bg-slate-900 border border-slate-800 rounded-xl p-2.5 text-xs text-slate-300 space-y-1">
                  <span className="font-bold text-emerald-400 flex items-center gap-1.5 text-xs">
                    <ThumbsUp className="w-4 h-4" />
                    Карта технологических допусков (АО «Полёт»):
                  </span>
                  <p className="text-[10px] text-slate-400 leading-snug">
                    Отклонения, разрешенные техническими условиями чертежа и стандартами ГОСТ 15467-79.
                  </p>
                </div>

                {report.detectedDefects?.filter(d => d.isTolerableFlaw || d.admissibility === 'ACCEPTABLE').map((d, i) => (
                  <div key={i} className="bg-slate-900 border border-emerald-500/30 rounded-xl p-2.5 text-xs space-y-1">
                    <div className="flex justify-between items-center">
                      <span className="font-bold text-slate-100 text-[11px]">{d.defectName}</span>
                      <span className="text-[9px] font-mono text-emerald-400 font-bold bg-emerald-950/60 px-2 py-0.2 rounded border border-emerald-800">
                        ДОПУСТИМО
                      </span>
                    </div>
                    <div className="text-[10px] text-slate-300 bg-slate-950 p-2 rounded-lg leading-snug">
                      <strong className="text-cyan-400 block mb-0.5 font-mono text-[9px]">Обоснование:</strong>
                      {d.toleranceRationale}
                    </div>
                    <div className="text-[9px] font-mono text-slate-400">Локализация: {d.exactLocation}</div>
                  </div>
                ))}
              </div>
            )}

            {/* ВКЛАДКА 3: Микропрофиль */}
            {activeTab === 'micro' && (
              <div className="bg-slate-900 border border-slate-800 rounded-xl p-3 text-xs space-y-2">
                <span className="font-bold text-cyan-400 flex items-center gap-1.5 text-xs">
                  <Gauge className="w-3.5 h-3.5" />
                  Топография шероховатости (ГОСТ 2789-73):
                </span>

                <div className="grid grid-cols-2 gap-1.5 pt-1 border-t border-slate-800">
                  <div className="bg-slate-950 p-2 rounded-lg">
                    <span className="text-[8px] text-slate-500 block font-mono">Норма чертежа:</span>
                    <span className="font-bold text-cyan-400 font-mono text-[11px]">{report.metrologicalPassport?.roughnessAudit?.drawingNorm}</span>
                  </div>
                  <div className="bg-slate-950 p-2 rounded-lg">
                    <span className="text-[8px] text-slate-500 block font-mono">Измеренное Ra (факт):</span>
                    <span className="font-bold text-amber-400 font-mono text-[11px]">{report.metrologicalPassport?.roughnessAudit?.measuredRaRange}</span>
                  </div>
                  <div className="bg-slate-950 p-2 rounded-lg">
                    <span className="text-[8px] text-slate-500 block font-mono">Диапазон Rz:</span>
                    <span className="font-mono text-slate-200 text-[10px]">{report.metrologicalPassport?.roughnessAudit?.estimatedRz}</span>
                  </div>
                  <div className="bg-slate-950 p-2 rounded-lg">
                    <span className="text-[8px] text-slate-500 block font-mono">Шаг неровностей Sm:</span>
                    <span className="font-mono text-slate-200 text-[10px]">{report.metrologicalPassport?.roughnessAudit?.stepSm || "0.08 - 0.25 мм"}</span>
                  </div>
                </div>

                <div className="bg-slate-950 p-2 rounded-lg text-[10px] space-y-1">
                  <div><span className="text-slate-500">Направление следов:</span> {report.metrologicalPassport?.roughnessAudit?.layDirection}</div>
                  <div><span className="text-slate-500">Статус соответствия:</span> <span className="text-slate-200 font-semibold">{report.metrologicalPassport?.roughnessAudit?.complianceStatus}</span></div>
                  <div><span className="text-slate-500">Базы (ГОСТ 21495):</span> {report.metrologicalPassport?.basesAndMatingSurfaces}</div>
                  <div><span className="text-slate-500">Фаски (ГОСТ 2.307):</span> {report.metrologicalPassport?.edgesAndChamfers}</div>
                </div>
              </div>
            )}

            {/* ВКЛАДКА 4: FMEA-риски */}
            {activeTab === 'fmea' && (
              <div className="flex flex-col gap-1.5">
                {report.fmeaOperationalRisks?.map((r, i) => (
                  <div key={i} className="bg-slate-900 border border-slate-800 rounded-xl p-2.5 text-xs space-y-1">
                    <div className="flex justify-between items-center">
                      <span className="font-bold text-white text-[11px]">{r.riskFactor}</span>
                      <span className={`text-[9px] font-mono font-bold px-1.5 py-0.2 rounded border ${
                        r.riskLevel === 'КРИТИЧЕСКИЙ' ? 'bg-rose-950 text-rose-400 border-rose-800' :
                        r.riskLevel === 'ВЫСОКИЙ' ? 'bg-orange-950 text-orange-400 border-orange-800' :
                        r.riskLevel === 'УМЕРЕННЫЙ' ? 'bg-amber-950 text-amber-400 border-amber-800' :
                        'bg-emerald-950 text-emerald-400 border-emerald-800'
                      }`}>{r.riskLevel}</span>
                    </div>
                    <p className="text-[10px] text-slate-400 bg-slate-950 p-1.5 rounded-lg">{r.impactDescription}</p>
                    <div className="text-[10px] text-emerald-300 bg-emerald-950/30 p-1.5 rounded border border-emerald-900/40">
                      <strong className="text-emerald-400 block text-[9px]">Оценка допустимых отклонений:</strong>
                      {r.toleratedEffect}
                    </div>
                  </div>
                ))}
              </div>
            )}

          </div>
        )}

      </main>

      {/* Закрепленная нижняя кнопка запуска под большой палец */}
      <div className="fixed bottom-0 inset-x-0 max-w-[440px] mx-auto bg-slate-950/95 backdrop-blur-xl border-t border-slate-800/80 p-2.5 z-40">
        <button
          onClick={() => runDeepMetrologicalAudit()}
          disabled={isAnalyzing}
          className={`w-full py-3 px-4 rounded-xl font-black text-xs uppercase tracking-wider flex items-center justify-center gap-2 shadow-lg transition active:scale-[0.98] ${
            isAnalyzing
              ? 'bg-slate-800 text-slate-500 cursor-not-allowed border border-slate-700'
              : isAutoMode
              ? 'bg-gradient-to-r from-cyan-400 via-sky-400 to-blue-500 text-slate-950 shadow-cyan-500/25'
              : 'bg-gradient-to-r from-slate-200 to-cyan-400 text-slate-950 shadow-cyan-500/20'
          }`}
        >
          {isAnalyzing ? (
            <>
              <RefreshCw className="w-4 h-4 animate-spin text-slate-950" />
              {isAutoMode ? "Автоопределение качества..." : "Прецизионный аудит..."}
            </>
          ) : (
            <>
              <Shield className="w-4 h-4 fill-slate-950" />
              {report 
                ? (isAutoMode ? "АВТО: ПОВТОРИТЬ ОЦЕНКУ" : "ПОВТОРИТЬ АУДИТ ПО ЧЕРТЕЖУ")
                : (isAutoMode ? "ДИМАН АВТО: ОЦЕНИТЬ ДЕТАЛЬ" : "ДИМАН: ВЫНЕСТИ РЕШЕНИЕ ПО СЕРИИ")
              }
            </>
          )}
        </button>
      </div>

    </div>
  );
}
